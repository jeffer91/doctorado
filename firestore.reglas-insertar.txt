// REGLAS DEL RESPALDO (Firebase es la base secundaria; la principal es Supabase).
// Copiar SOLO este match dentro de:
// service cloud.firestore { match /databases/{database}/documents { ... } }
// NO reemplazar las demás reglas del proyecto UTET.
match /doctorado_respuestas/{respuestaId} {
  function esAdminDoctorado() {
    return request.auth != null
      && request.auth.token.email_verified == true
      && request.auth.token.email == "jeffersonvillarreal91@gmail.com"
      && request.auth.token.firebase.sign_in_provider == "google.com";
  }

  function escala(v) { return v == null || (v is int && v >= 1 && v <= 5); }

  function datosValidos() {
    let a = request.resource.data;
    let r = a.r;
    return a.keys().hasOnly(["etapa", "cedula", "r", "consent", "schemaVersion", "anonUid", "createdAt"])
      && a.keys().hasAll(["etapa", "cedula", "r", "consent", "schemaVersion", "anonUid", "createdAt"])
      && respuestaId.matches("^[a-f0-9]{64}$")
      && a.etapa in ["inicio", "proceso", "final"]
      && a.cedula is string && a.cedula.matches("^[0-9]{10}$")
      && a.consent == true
      && a.schemaVersion == 2
      && a.anonUid == request.auth.uid
      && a.createdAt == request.time
      && r is map
      && r.keys().hasOnly(["se1","se2","se3","se4","se5","p6","p7","horas","horas_ns","dificultad","dificultad_otra","ia","omitidas"])
      && r.keys().hasAll(["se1","se2","se3","se4","se5","p6","p7","horas","horas_ns","dificultad","dificultad_otra","ia","omitidas"])
      && escala(r.se1) && escala(r.se2) && escala(r.se3) && escala(r.se4) && escala(r.se5)
      && escala(r.p6) && escala(r.p7) && escala(r.ia)
      && (r.horas == null || (r.horas is number && r.horas >= 0 && r.horas <= 100))
      && r.horas_ns is bool
      && (r.dificultad == null || r.dificultad in ["tema","bibliografia","metodologia","analisis","redaccion","tiempo","tecnologia","otra"])
      && (r.dificultad_otra == null || (r.dificultad_otra is string && r.dificultad_otra.size() <= 200))
      && r.omitidas is list && r.omitidas.size() <= 10;
  }

  allow get, list: if esAdminDoctorado();
  allow create: if request.auth != null
    && request.auth.token.firebase.sign_in_provider == "anonymous"
    && datosValidos();
  // Sin actualización ni eliminación: protege el registro original.
  allow update, delete: if false;
}
