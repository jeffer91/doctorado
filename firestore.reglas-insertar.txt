// El requisito Google + email verificado se ejecuta en el servidor Firestore.
match /doctorado_respuestas/{respuestaId} {
  function esAdminDoctorado() {
    return request.auth != null
      && request.auth.token.email_verified == true
      && request.auth.token.email == "jeffersonvillarreal91@gmail.com"
      && request.auth.token.firebase.sign_in_provider == "google.com";
  }

  function esParticipanteAnonimo() {
    return request.auth != null
      && request.auth.token.firebase.sign_in_provider == "anonymous";
  }

  function datosValidos() {
    let a = request.resource.data;
    return a.keys().hasOnly([
        "etapa", "cedula", "carrera", "answers",
        "consent", "schemaVersion", "anonUid", "createdAt"
      ])
      && a.keys().hasAll([
        "etapa", "cedula", "carrera", "answers",
        "consent", "schemaVersion", "anonUid", "createdAt"
      ])
      && respuestaId.matches("^[a-f0-9]{64}$")
      && a.etapa in ["inicio", "desarrollo", "finalizados"]
      && a.cedula is string
      && a.cedula.matches("^[0-9]{10}$")
      && a.carrera is string && a.carrera.size() >= 3 && a.carrera.size() <= 120
      && a.consent == true
      && a.schemaVersion == 1
      && a.anonUid == request.auth.uid
      && a.createdAt == request.time
      && a.answers is map
      && a.answers.keys().hasOnly(["q1","q2","q3","q4","q5","q6","q7","q8","q9","q10"])
      && a.answers.keys().hasAll(["q1","q2","q3","q4","q5","q6","q7","q8","q9","q10"])
      && a.answers.q1 is list && a.answers.q1.size() > 0 && a.answers.q1.size() <= 6
      && a.answers.q9 is list && a.answers.q9.size() > 0 && a.answers.q9.size() <= 2
      && a.answers.q10 is string && a.answers.q10.size() <= 600
      && (
        (a.etapa in ["inicio","desarrollo"]
          && a.answers.q4 is map
          && a.answers.q4.keys().hasAll(["problema","fuentes","metodos","datos","redaccion"]))
        ||
        (a.etapa == "finalizados"
          && a.answers.q3 is map
          && a.answers.q3.keys().hasAll(["problema","fuentes","metodos","datos","redaccion"]))
      );
  }

  allow get, list: if esAdminDoctorado();
  allow create: if esParticipanteAnonimo() && datosValidos();
  // Sin actualización ni eliminación: protege el registro original de investigación.
  allow update, delete: if false;
}
